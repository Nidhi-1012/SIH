function findRoutes() {
  const destInput = document.getElementById('desk-dest').value.trim() || document.getElementById('mob-dest').value.trim();
  if (!destInput) {
    showToast('Please type or select a destination');
    return;
  }

  showToast('Analyzing safest routes via AI Engine...');

  const key = Object.keys(DB).find(k => destInput.toLowerCase().includes(k)) || 'shillong';
  const localData = DB[key];

  try {
    const res = await fetch('/api/v1/route', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ origin: 'Guwahati', destination: destInput, priority_class: 'P2' })
    }).catch(() => null);

    if (res && res.ok) {
      const data = await res.json();
      const routesList = Array.isArray(data) ? data : (data.routes || []);

      if (routesList.length > 0) {
        const mergedRoutes = routesList.map((rt, i) => {
          const fallback = localData ? localData.routes[i % localData.routes.length] : null;
          const pts = (rt.geometry_points && rt.geometry_points.length > 0) ? rt.geometry_points : (fallback ? fallback.pts : []);
          return {
            pts: pts,
            overall: Math.round(rt.overall_risk_score || rt.risk_score || (fallback ? fallback.overall : 22)),
            ls: Math.round(rt.score_breakdown?.landslide_risk || (fallback ? fallback.ls : 14)),
            rain: Math.round(rt.score_breakdown?.weather_risk || (fallback ? fallback.rain : 12)),
            flood: Math.round(rt.score_breakdown?.flood_risk || (fallback ? fallback.flood : 4)),
            dist: rt.distance_km ? `${rt.distance_km} km` : (fallback ? fallback.dist : '92 km'),
            time: rt.eta_minutes ? `${Math.floor(rt.eta_minutes / 60)}h ${Math.round(rt.eta_minutes % 60)}m` : (fallback ? fallback.time : '2h 10m'),
            via: rt.name || (fallback ? fallback.via : 'Safe Highway'),
            road: 'Good NH Standard',
            id: rt.route_id || `r-${i}`
          };
        });
        renderInlineRoutes(mergedRoutes, destInput);
        return;
      }
    }
  } catch (e) {
    console.warn('Backend routing fallback to verified pilot graph', e);
  }

  // Fallback to local verified routes
  renderInlineRoutes(localData.routes, destInput);
}
